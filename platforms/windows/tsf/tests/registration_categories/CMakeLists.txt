cmake_minimum_required(VERSION 3.25)
project(TsfCategoryRegistrationRegression LANGUAGES CXX)
enable_testing()
set(registration_source "${CMAKE_CURRENT_SOURCE_DIR}/../../Register/Register.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${registration_source}")
file(READ "${registration_source}" registration)
string(FIND "${registration}" "BOOL RegisterCategories()" begin)
string(FIND "${registration}" "LONG RecurseDeleteKey(" end)
if(begin LESS 0 OR end LESS_EQUAL begin)
  message(FATAL_ERROR "Could not locate production category registration bodies")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${registration}" ${begin} ${length} category_body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registration_categories.inc" "${category_body}")
add_executable(msime-tsf-category-registration-test registration_categories.cpp)
target_include_directories(msime-tsf-category-registration-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(msime-tsf-category-registration-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(msime-tsf-category-registration-test PRIVATE /W4 /WX)
else()
  target_compile_options(msime-tsf-category-registration-test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME msime-tsf-category-registration COMMAND msime-tsf-category-registration-test)

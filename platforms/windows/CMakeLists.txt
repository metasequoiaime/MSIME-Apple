cmake_minimum_required(VERSION 3.25)
project(MSIMEClientWindowsBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
add_executable(windows-registration-inbox tests/registration_inbox.cpp)
target_include_directories(windows-registration-inbox PRIVATE .)
target_link_libraries(windows-registration-inbox PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(windows-registration-inbox PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(windows-registration-inbox PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME windows-registration-inbox COMMAND windows-registration-inbox)
set_tests_properties(windows-registration-inbox PROPERTIES TIMEOUT 10)
add_executable(windows-focus-router tests/focus_router.cpp)
target_include_directories(windows-focus-router PRIVATE . ../../vendor/MSIME-Engine/contracts)
target_link_libraries(windows-focus-router PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(windows-focus-router PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(windows-focus-router PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME windows-focus-router COMMAND windows-focus-router)
set_tests_properties(windows-focus-router PROPERTIES TIMEOUT 10)
add_executable(windows-main-frame tests/main_frame.cpp)
target_include_directories(windows-main-frame PRIVATE . ../../vendor/MSIME-Engine/contracts)
if(MSVC)
  target_compile_options(windows-main-frame PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(windows-main-frame PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME windows-main-frame COMMAND windows-main-frame)
add_executable(windows-focus-gate tests/focus_gate.cpp)
target_include_directories(windows-focus-gate PRIVATE .)
target_link_libraries(windows-focus-gate PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(windows-focus-gate PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(windows-focus-gate PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME windows-focus-gate COMMAND windows-focus-gate)
set_tests_properties(windows-focus-gate PROPERTIES TIMEOUT 10)
add_library(msime-windows-replies STATIC ReplyCodec.cpp)
target_include_directories(msime-windows-replies PUBLIC . ../../vendor/MSIME-Engine/contracts)
if(MSVC)
  target_compile_options(msime-windows-replies PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(msime-windows-replies PRIVATE -Wall -Wextra -Werror)
endif()
option(MSIME_WINDOWS_PIPE_ONLY "Build native pipe tests without the Rust host library" OFF)
if(WIN32)
  add_subdirectory(tests/native-pipe)
endif()
if(MSIME_WINDOWS_PIPE_ONLY)
  if(NOT WIN32)
    message(FATAL_ERROR "Pipe-only tests require a Windows target")
  endif()
  return()
endif()
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
set(MSIME_HOST_LIBRARY "" CACHE FILEPATH "Built shared host library or Windows DLL import library")
if(NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "Set MSIME_HOST_LIBRARY to the Cargo-built shared host library for this target")
endif()
add_library(msime-windows-session STATIC ServerSession.cpp)
target_include_directories(msime-windows-session PUBLIC . ../../crates/host-api/include ../../vendor/MSIME-Engine/contracts)
target_link_libraries(msime-windows-session PUBLIC nlohmann_json::nlohmann_json Threads::Threads "${MSIME_HOST_LIBRARY}")
enable_testing()
add_executable(windows-session-smoke tests/session_smoke.cpp tests/session_pump.cpp tests/session_workers.cpp)
add_executable(windows-reply-codec tests/reply_codec.cpp)
target_link_libraries(windows-reply-codec PRIVATE msime-windows-replies)
add_library(msime-windows-composer STATIC ReplyComposer.cpp)
target_link_libraries(msime-windows-composer PUBLIC msime-windows-session msime-windows-replies)
add_library(msime-windows-focused STATIC FocusedSession.cpp)
target_link_libraries(msime-windows-focused PUBLIC msime-windows-composer)
add_library(msime-windows-input STATIC InputQueue.cpp)
target_link_libraries(msime-windows-input PUBLIC msime-windows-focused)
add_library(msime-windows-pump STATIC SessionPump.cpp SessionWorkers.cpp SessionController.cpp)
target_link_libraries(msime-windows-pump PUBLIC msime-windows-input)
target_link_libraries(windows-session-smoke PRIVATE msime-windows-pump)
add_executable(windows-input-queue tests/input_queue.cpp)
target_link_libraries(windows-input-queue PRIVATE msime-windows-input)
add_test(NAME windows-input-queue COMMAND windows-input-queue)
set_tests_properties(windows-input-queue PROPERTIES TIMEOUT 30)
add_executable(windows-reply-composer tests/reply_composer.cpp)
target_link_libraries(windows-reply-composer PRIVATE msime-windows-composer)
foreach(target msime-windows-session windows-session-smoke windows-reply-codec msime-windows-composer windows-reply-composer msime-windows-focused msime-windows-input windows-input-queue msime-windows-pump)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(WIN32)
  add_library(msime-windows-server STATIC WindowsServer.cpp)
  target_link_libraries(msime-windows-server PUBLIC msime-windows-pump msime-windows-pipe)
  add_executable(windows-server-smoke tests/server_smoke.cpp)
  target_link_libraries(windows-server-smoke PRIVATE msime-windows-server)
  foreach(target msime-windows-server windows-server-smoke)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
  add_test(NAME windows-server COMMAND windows-server-smoke)
  set_tests_properties(windows-server PROPERTIES TIMEOUT 30)
endif()

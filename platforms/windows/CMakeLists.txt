cmake_minimum_required(VERSION 3.25)
project(MSIMEClientWindowsBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
set(MSIME_HOST_LIBRARY "" CACHE FILEPATH "Built shared host library or Windows DLL import library")
if(NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "Set MSIME_HOST_LIBRARY to the Cargo-built shared host library for this target")
endif()
add_library(msime-windows-session STATIC ServerSession.cpp)
target_include_directories(msime-windows-session PUBLIC . ../../crates/host-api/include ../../vendor/MSIME-Engine/contracts)
target_link_libraries(msime-windows-session PUBLIC nlohmann_json::nlohmann_json Threads::Threads "${MSIME_HOST_LIBRARY}")
enable_testing()
add_executable(windows-session-smoke tests/session_smoke.cpp)
target_link_libraries(windows-session-smoke PRIVATE msime-windows-session)
foreach(target msime-windows-session windows-session-smoke)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
add_test(NAME windows-session COMMAND windows-session-smoke)

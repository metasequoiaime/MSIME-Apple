cmake_minimum_required(VERSION 3.25)
project(MSIMEClientLinux LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0>=1.5.20)
find_package(nlohmann_json 3.11 REQUIRED)
option(MSIME_LINUX_VOICE "Build the shared Voice library and Linux contract test" OFF)
set(MSIME_HOST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/debug/libmsime_host_api.so" CACHE FILEPATH "Cargo-built Linux host library")
set(MSIME_RUNTIME_OPTIONS_FILE "" CACHE FILEPATH
    "Prepared runtime-options.json to install for the IBus component")
if(NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "Build msime-host-api for Linux first")
endif()
add_library(msime-ibus-host STATIC ClientEngine.cpp)
target_compile_options(msime-ibus-host PRIVATE -Wall -Wextra -Werror)
target_include_directories(msime-ibus-host PUBLIC . ../../crates/host-api/include)
target_link_libraries(msime-ibus-host PUBLIC PkgConfig::IBUS nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}")
add_executable(msime-client-ibus main.cpp)
target_compile_options(msime-client-ibus PRIVATE -Wall -Wextra -Werror)
target_link_libraries(msime-client-ibus PRIVATE msime-ibus-host)
add_executable(ibus-engine-smoke tests/engine_smoke.cpp)
target_compile_options(ibus-engine-smoke PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ibus-engine-smoke PRIVATE msime-ibus-host)
add_executable(msime-client-dictionary dictionary_request.cpp)
target_compile_options(msime-client-dictionary PRIVATE -Wall -Wextra -Werror)
target_include_directories(msime-client-dictionary PRIVATE ../../crates/host-api/include)
target_link_libraries(msime-client-dictionary PRIVATE nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}")
add_executable(msime-client-clipboard clipboard_history.cpp)
target_compile_options(msime-client-clipboard PRIVATE -Wall -Wextra -Werror)
target_link_libraries(msime-client-clipboard PRIVATE nlohmann_json::nlohmann_json)
add_executable(msime-client-online-provider-contract tests/online_provider_contract.cpp)
target_compile_options(msime-client-online-provider-contract PRIVATE -Wall -Wextra -Werror)
target_include_directories(msime-client-online-provider-contract PRIVATE ../../crates/host-api/include)
target_link_libraries(msime-client-online-provider-contract PRIVATE nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}")
add_test(NAME linux_online_provider_contract COMMAND msime-client-online-provider-contract)
if(MSIME_LINUX_VOICE)
  set(MSIME_VOICE_CAPTURE OFF CACHE BOOL "" FORCE)
  set(MSIME_VOICE_WHISPER OFF CACHE BOOL "" FORCE)
  add_subdirectory(../../vendor/MSIME-Engine/voice msime-engine-voice EXCLUDE_FROM_ALL)
  add_executable(msime-client-voice-contract tests/voice_contract.cpp)
  target_compile_options(msime-client-voice-contract PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(msime-client-voice-contract PRIVATE MetasequoiaIme::Voice)
  add_test(NAME linux_voice_contract COMMAND msime-client-voice-contract)
endif()
configure_file(msime-client-preview.xml.in msime-client-preview.xml @ONLY)
install(TARGETS msime-client-ibus msime-client-dictionary msime-client-clipboard
        RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/msime-client-preview.xml"
        DESTINATION "${CMAKE_INSTALL_DATADIR}/ibus/component")
if(MSIME_RUNTIME_OPTIONS_FILE)
  if(NOT EXISTS "${MSIME_RUNTIME_OPTIONS_FILE}")
    message(FATAL_ERROR "MSIME_RUNTIME_OPTIONS_FILE does not exist")
  endif()
  install(FILES "${MSIME_RUNTIME_OPTIONS_FILE}"
          DESTINATION "${CMAKE_INSTALL_SYSCONFDIR}/msime-client"
          RENAME runtime-options.json)
endif()

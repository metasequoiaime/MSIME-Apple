cmake_minimum_required(VERSION 3.25)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version; match the Cargo build")
project(MSIMEClientMacOS LANGUAGES CXX OBJCXX)
option(MSIME_MACOS_VOICE_CAPTURE "Build the shared macOS voice capture adapter" OFF)
option(MSIME_MACOS_VOICE_SERVICE "Build the Apple voice input service" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(MSIME_HOST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/debug/libmsime_host_api.a" CACHE FILEPATH "Rust host static library for this architecture")
if(NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "Build msime-host-api with Cargo first")
endif()
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(SQLITE_TARGET SQLite3::SQLite3)
else()
  set(SQLITE_TARGET SQLite::SQLite3)
endif()
add_library(apple-client STATIC ../../shared/apple/MSIMEClientSession.mm ../../shared/apple/TextClient.mm)
target_compile_options(apple-client PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_include_directories(apple-client PUBLIC ../../shared/apple . ../../crates/host-api/include)
target_link_libraries(apple-client PUBLIC "${MSIME_HOST_LIBRARY}" ${SQLITE_TARGET} "-framework Foundation" "-framework Security")
add_executable(MSIMEClientInputMethod MACOSX_BUNDLE main.mm InputSourceRegistration.mm PreferencesWindowController.mm InputController.mm AppearancePreferences.mm CandidateSkin.cpp CandidateSkinPreviewView.mm SkinSettingsView.mm)
target_compile_options(MSIMEClientInputMethod PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(MSIMEClientInputMethod PRIVATE apple-client "-framework AppKit" "-framework InputMethodKit" "-framework Carbon")
set_target_properties(MSIMEClientInputMethod PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
set(MSIME_OPTIONS_FILE "" CACHE FILEPATH "Optional development runtime configuration; never distribute this file")
if(MSIME_MACOS_VOICE_CAPTURE AND NOT MSIME_MACOS_VOICE_SERVICE)
  set(MSIME_VOICE_CAPTURE ON CACHE BOOL "" FORCE)
  set(MSIME_VOICE_WHISPER OFF CACHE BOOL "" FORCE)
  set(MSIME_VOICE_WINDOWS_APP OFF CACHE BOOL "" FORCE)
  set(MSIME_VOICE_APPLE_EXAMPLE OFF CACHE BOOL "" FORCE)
  add_subdirectory(../../vendor/MSIME-Engine/voice ${CMAKE_BINARY_DIR}/engine-voice)
  target_link_libraries(MSIMEClientInputMethod PRIVATE MetasequoiaIme::VoiceCapture)
endif()
if(MSIME_MACOS_VOICE_SERVICE)
  set(MSIME_VOICE_CAPTURE ON CACHE BOOL "" FORCE)
  set(MSIME_VOICE_WHISPER ON CACHE BOOL "" FORCE)
  set(MSIME_VOICE_WINDOWS_APP OFF CACHE BOOL "" FORCE)
  set(MSIME_VOICE_APPLE_EXAMPLE OFF CACHE BOOL "" FORCE)
  add_subdirectory(../../vendor/MSIME-Engine/voice ${CMAKE_BINARY_DIR}/engine-voice-service)
  add_library(voice-input-service STATIC VoiceInputService.mm)
  target_include_directories(voice-input-service PRIVATE ../../vendor/MSIME-Engine/voice/include)
  target_compile_options(voice-input-service PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(voice-input-service PRIVATE voice-settings MetasequoiaIme::VoiceCapture MetasequoiaIme::VoiceWhisper MetasequoiaIme::Voice "-framework AppKit" "-framework AVFoundation")
endif()
if(MSIME_OPTIONS_FILE)
  target_sources(MSIMEClientInputMethod PRIVATE "${MSIME_OPTIONS_FILE}")
  set_source_files_properties("${MSIME_OPTIONS_FILE}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
